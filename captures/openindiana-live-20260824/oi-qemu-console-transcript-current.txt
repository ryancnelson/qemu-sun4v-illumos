                FSTYP=$(/usr/lib/fs/ufs/fstyp $rdev 2>/dev/null)
        fi
        if [ $? -eq 0 ]; then
                $MOUNT -F $FSTYP -o $OPTIONS $dev /.cdrom || break
                [ -f /.cdrom/.volsetid ] && cdvolsetid=$( < "/.cdrom/.volsetid"
)
                if [ "$volsetid" != "$cdvolsetid" ]; then
                        /sbin/umount -f $dev
                        continue
                fi
                if [ -f $SOLARIS_ZLIB ]; then
                        echo "" > /.liveusb
                        break
                else
                        /sbin/umount -f $dev
                        continue
                fi
        fi

        # For systems that might have multiple CD drives,
        # make sure that the filesystem gets mounted off of
        # the same CD drive as the one we're booting from.
        #
        /usr/lib/fs/hsfs/fstyp -v $rdev 2>&1 | \
            grep "Volume set id: $volsetid" > /dev/null

        fi

        RETRIES=0
        until
                ROOTPATH=$(/sbin/dhcpinfo Rootpath)
        do
                RETRIES=$((RETRIES + 1))
                ((RETRIES > 10)) && break
                sleep 6
        done
        # Get scheme
        SCHEME=${ROOTPATH%%://*}
        [ "$SCHEME" = "$ROOTPATH" ] && SCHEME=
        # Handle nfs://<ip-address>/<path> form
        if [ "$SCHEME" = "nfs" ] ; then
                ROOTPATH=${ROOTPATH#nfs://}
                IPADDRESS=${ROOTPATH%%/*}
                ROOTPATH=${ROOTPATH#$IPADDRESS}
        fi
        # Handle <ip-address>:/<path> form
        if [ -z "$SCHEME" -a "${ROOTPATH/:}" != "$ROOTPATH" ] ; then
                SCHEME="nfs"
                IPADDRESS=${ROOTPATH%%:*}
                ROOTPATH=${ROOTPATH#$IPADDRESS:}
        fi
        # Try to mount

                # the nfs/mapid service

OPTIONS="ro,nologging,noatime"

/sbin/listusb | while read dev rdev

        FSTYP=$(/usr/lib/fs/hsfs/fstyp $rdev 2>/dev/null)
        if [ X$FSTYP != "Xhsfs" ]; then
                FSTYP=$(/usr/lib/fs/ufs/fstyp $rdev 2>/dev/null)
        fi
        if [ $? -eq 0 ]; then
                $MOUNT -F $FSTYP -o $OPTIONS $dev /.cdrom || break
                [ -f /.cdrom/.volsetid ] && cdvolsetid=$( < "/.cdrom/.volsetid"

        RETRIES=0
        until
                ROOTPATH=$(/sbin/dhcpinfo Rootpath)
        do
                RETRIES=$((RETRIES + 1))
                ((RETRIES > 10)) && break
                sleep 6
        done
        # Get scheme
        SCHEME=${ROOTPATH%%://*}
        [ "$SCHEME" = "$ROOTPATH" ] && SCHEME=
        # Handle nfs://<ip-address>/<path> form
        if [ "$SCHEME" = "nfs" ] ; then
                ROOTPATH=${ROOTPATH#nfs://}
                IPADDRESS=${ROOTPATH%%/*}
                ROOTPATH=${ROOTPATH#$IPADDRESS}

        # Handle <ip-address>:/<path> form
        if [ -z "$SCHEME" -a "${ROOTPATH/:}" != "$ROOTPATH" ] ; then
                SCHEME="nfs"

                ROOTPATH=${ROOTPATH#$IPADDRESS:}

$MOUNT -F hsfs -o ro ${misc_lofi_dev} /mnt/misc
